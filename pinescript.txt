// ─── ZigZag ───────────────────────────────────────────────────────────────────
f_is_sh(len) =>
    bool ok = true
    for j = 1 to len
        if high[len] <= high[len - j] or high[len] <= high[len + j]
            ok := false
            break
    ok

f_is_sl(len) =>
    bool ok = true
    for j = 1 to len
        if low[len] >= low[len - j] or low[len] >= low[len + j]
            ok := false
            break
    ok

is_sh   = f_is_sh(i_swing)
is_sl   = f_is_sl(i_swing)
pvt_bar = bar_index - i_swing

zz_n()      => array.size(zp)

zz_last_p() => zz_n() > 0 ? array.get(zp, zz_n() - 1) : float(na)

zz_push(p, b, t) =>
    array.push(zp, p)
    array.push(zb, b)
    array.push(zt, t)
    if array.size(zp) > 50

zz_replace(p, b) =>
    n = zz_n()
    if n > 0
        array.set(zp, n - 1, p)
        array.set(zb, n - 1, b)

// ─── BC→CD lookup (_BC_TO_CD tablosundan en yakın cd_r) ───────────────────────
// keys: 0.382, 0.500, 0.618, 0.786, 0.886
// Her pattern için cd değerleri basari.py _BC_TO_CD tablosundan
f_nearest_cd(ab_bc, cd382, cd500, cd618, cd786, cd886) =>
    float best_cd = cd618
    float best_d  = 999.0
    d382 = math.abs(ab_bc - 0.382)
    d500 = math.abs(ab_bc - 0.500)
    d618 = math.abs(ab_bc - 0.618)
    d786 = math.abs(ab_bc - 0.786)
    d886 = math.abs(ab_bc - 0.886)
    if d382 < best_d
        best_d  := d382
        best_cd := cd382
    if d500 < best_d
        best_d  := d500
        best_cd := cd500
    if d618 < best_d
        best_d  := d618
        best_cd := cd618
    if d786 < best_d
        best_d  := d786
        best_cd := cd786
    if d886 < best_d
        best_cd := cd886
    best_cd

f_in(v, lo, hi) => v >= lo and v <= hi

f_conf(xa_ab, xa_lo, xa_hi, ab_bc, ab_lo, ab_hi) =>
    xa_dev = math.abs(xa_ab - (xa_lo + xa_hi) / 2.0) / math.max((xa_lo + xa_hi) / 2.0, 1e-10)
    ab_dev = math.abs(ab_bc - (ab_lo + ab_hi) / 2.0) / math.max((ab_lo + ab_hi) / 2.0, 1e-10)
    math.max(0.0, math.min(100.0, 100.0 - (xa_dev * 40.0 + ab_dev * 40.0)))

// ─── PRZ builder: d1=XA*AD proj, d2=BC→CD lookup, dar kutu ───────────────────
f_prz(d1, d2, x_p, bullish, retrace) =>
    // X-clamp: retracement patternlerde D, X'i geçemez
    float lo = math.min(d1, d2)
    float hi = math.max(d1, d2)
    if retrace
        if bullish
            lo := math.max(lo, x_p)
            hi := math.max(hi, x_p)
        else
            lo := math.min(lo, x_p)
            hi := math.min(hi, x_p)
    // Çok geniş PRZ reddet (>%4)
    ctr = math.abs((lo + hi) / 2.0)
    gap_pct = ctr > 1e-10 ? (hi - lo) / ctr * 100 : 99.0
    valid = gap_pct <= 4.0
    [lo, hi, valid]

f_ddist(cur_p, a_p, xa, ad_lo, ad_hi, d1, bullish) =>
    dlo = bullish ? a_p - xa * ad_hi : a_p + xa * ad_lo
    dhi = bullish ? a_p - xa * ad_lo : a_p + xa * ad_hi
    cur_p >= math.min(dlo, dhi) and cur_p <= math.max(dlo, dhi) ? 0.0 : math.abs(cur_p - d1) / math.max(math.abs(d1), 1e-10) * 100

f_check(x_p, a_p, b_p, c_p, bullish, cur_p) =>
    xa   = math.abs(x_p - a_p)
    ab   = math.abs(a_p - b_p)
    bc   = math.abs(b_p - c_p)
    bool   found = false
    string pname = ""
    float  pconf = 0.0
    float  plo   = na
    float  phi   = na

    // Yapı kontrolü: alternating high/low
    struct_ok = false
    if bullish
        struct_ok := b_p > x_p and b_p < a_p and c_p > b_p and c_p < a_p
    else
        struct_ok := b_p < x_p and b_p > a_p and c_p < b_p and c_p > a_p

    if xa > 1e-10 and ab > 1e-10 and struct_ok
        xa_ab  = ab / xa
        ab_bc  = bc / math.max(ab, 1e-10)
        xb_ext = math.abs(x_p - b_p) / xa
        xc_prj = math.abs(x_p - c_p) / xa

        // d1 = XA * xa_ad_mid (primary projection)
        // d2 = BC * cd_r      (BC→CD lookup)

        // GARTLEY — _BC_TO_CD: {0.382:1.618, 0.500:1.414, 0.618:1.272, 0.786:1.272, 0.886:1.272}
        if i_gartley and not found
            if f_in(xa_ab, 0.550, 0.680) and f_in(ab_bc, 0.382, 0.886)
                cv = f_conf(xa_ab, 0.550, 0.680, ab_bc, 0.382, 0.886)
                if cv >= i_conf
                    xa_ad_mid = (0.750 + 0.820) / 2.0
                    d1  = bullish ? a_p - xa * xa_ad_mid : a_p + xa * xa_ad_mid
                    cd_r = f_nearest_cd(ab_bc, 1.618, 1.414, 1.272, 1.272, 1.272)
                    d2  = bullish ? c_p - bc * cd_r : c_p + bc * cd_r
                    dd  = f_ddist(cur_p, a_p, xa, 0.750, 0.820, d1, bullish)
                    if dd <= i_ddist
                        [cl, ch, ok] = f_prz(d1, d2, x_p, bullish, true)
                        if ok
                            found := true
                            pname := "Gartley"
                            pconf := cv
                            plo   := cl
                            phi   := ch

        // BAT — _BC_TO_CD: {0.382:2.618, 0.500:2.000, 0.618:1.618, 0.786:1.618, 0.886:1.618}
        if i_bat and not found
            if f_in(xa_ab, 0.320, 0.560) and f_in(ab_bc, 0.382, 0.886)
                cv = f_conf(xa_ab, 0.320, 0.560, ab_bc, 0.382, 0.886)
                if cv >= i_conf
                    xa_ad_mid = (0.850 + 0.920) / 2.0
                    d1  = bullish ? a_p - xa * xa_ad_mid : a_p + xa * xa_ad_mid
                    cd_r = f_nearest_cd(ab_bc, 2.618, 2.000, 1.618, 1.618, 1.618)
                    d2  = bullish ? c_p - bc * cd_r : c_p + bc * cd_r
                    dd  = f_ddist(cur_p, a_p, xa, 0.850, 0.920, d1, bullish)
                    if dd <= i_ddist
                        [cl, ch, ok] = f_prz(d1, d2, x_p, bullish, true)
                        if ok
                            found := true
                            pname := "Bat"
                            pconf := cv
                            plo   := cl
                            phi   := ch

        // ALTBAT — _BC_TO_CD: {0.382:3.618, 0.500:2.618, 0.618:2.240, 0.786:2.000, 0.886:2.000}
        if i_altbat and not found
            if f_in(xa_ab, 0.080, 0.420) and f_in(ab_bc, 0.382, 0.886)
                cv = f_conf(xa_ab, 0.080, 0.420, ab_bc, 0.382, 0.886)
                if cv >= i_conf
                    xa_ad_mid = (1.080 + 1.180) / 2.0
                    d1  = bullish ? a_p - xa * xa_ad_mid : a_p + xa * xa_ad_mid
                    cd_r = f_nearest_cd(ab_bc, 3.618, 2.618, 2.240, 2.000, 2.000)
                    d2  = bullish ? c_p - bc * cd_r : c_p + bc * cd_r
                    dd  = f_ddist(cur_p, a_p, xa, 1.080, 1.180, d1, bullish)
                    if dd <= i_ddist
                        [cl, ch, ok] = f_prz(d1, d2, x_p, bullish, false)
                        if ok
                            found := true
                            pname := "AltBat"
                            pconf := cv
                            plo   := cl
                            phi   := ch

        // BUTTERFLY — _BC_TO_CD: {0.382:2.618, 0.500:2.000, 0.618:2.240, 0.786:2.618, 0.886:2.618}
        if i_bfly and not found
            if f_in(xa_ab, 0.720, 0.850) and f_in(ab_bc, 0.382, 0.886)
                cv = f_conf(xa_ab, 0.720, 0.850, ab_bc, 0.382, 0.886)
                if cv >= i_conf
                    xa_ad_mid = (1.200 + 1.680) / 2.0
                    d1  = bullish ? a_p - xa * xa_ad_mid : a_p + xa * xa_ad_mid
                    cd_r = f_nearest_cd(ab_bc, 2.618, 2.000, 2.240, 2.618, 2.618)
                    d2  = bullish ? c_p - bc * cd_r : c_p + bc * cd_r
                    dd  = f_ddist(cur_p, a_p, xa, 1.200, 1.680, d1, bullish)
                    if dd <= i_ddist
                        [cl, ch, ok] = f_prz(d1, d2, x_p, bullish, false)
                        if ok
                            found := true
                            pname := "Butterfly"
                            pconf := cv
                            plo   := cl
                            phi   := ch

        // CRAB — _BC_TO_CD: {0.382:3.618, 0.500:2.618, 0.618:2.618, 0.786:2.618, 0.886:2.618}
        if i_crab and not found
            if f_in(xa_ab, 0.350, 0.650) and f_in(ab_bc, 0.382, 0.886)
                cv = f_conf(xa_ab, 0.350, 0.650, ab_bc, 0.382, 0.886)
                if cv >= i_conf
                    xa_ad_mid = (1.550 + 1.700) / 2.0
                    d1  = bullish ? a_p - xa * xa_ad_mid : a_p + xa * xa_ad_mid
                    cd_r = f_nearest_cd(ab_bc, 3.618, 2.618, 2.618, 2.618, 2.618)
                    d2  = bullish ? c_p - bc * cd_r : c_p + bc * cd_r
                    dd  = f_ddist(cur_p, a_p, xa, 1.550, 1.700, d1, bullish)
                    if dd <= i_ddist
                        [cl, ch, ok] = f_prz(d1, d2, x_p, bullish, false)
                        if ok
                            found := true
                            pname := "Crab"
                            pconf := cv
                            plo   := cl
                            phi   := ch

        // DEEPCRAB — _BC_TO_CD: {0.382:3.618, 0.500:2.618, 0.618:2.618, 0.786:2.618, 0.886:2.618}
        if i_dcrab and not found
            if f_in(xa_ab, 0.850, 0.930) and f_in(ab_bc, 0.382, 0.886)
                cv = f_conf(xa_ab, 0.850, 0.930, ab_bc, 0.382, 0.886)
                if cv >= i_conf
                    xa_ad_mid = (1.550 + 1.700) / 2.0
                    d1  = bullish ? a_p - xa * xa_ad_mid : a_p + xa * xa_ad_mid
                    cd_r = f_nearest_cd(ab_bc, 3.618, 2.618, 2.618, 2.618, 2.618)
                    d2  = bullish ? c_p - bc * cd_r : c_p + bc * cd_r
                    dd  = f_ddist(cur_p, a_p, xa, 1.550, 1.700, d1, bullish)
                    if dd <= i_ddist
                        [cl, ch, ok] = f_prz(d1, d2, x_p, bullish, false)
                        if ok
                            found := true
                            pname := "DeepCrab"
                            pconf := cv
                            plo   := cl
                            phi   := ch

        // SHARK — _BC_TO_CD: {0.382:2.240, 0.500:2.000, 0.618:1.618, 0.786:1.618, 0.886:1.618}
        if i_shark and not found
            shark_b_ok = bullish ? b_p > a_p : b_p < a_p
            if shark_b_ok and f_in(xb_ext, 1.080, 1.680) and f_in(ab_bc, 1.500, 2.350)
                xb_id  = (1.080 + 1.680) / 2.0
                ab_id  = (1.500 + 2.350) / 2.0
                xb_dev = math.abs(xb_ext - xb_id) / math.max(xb_id, 1e-10)
                ab_dev = math.abs(ab_bc  - ab_id) / math.max(ab_id, 1e-10)
                cv = math.max(0.0, math.min(100.0, 100.0 - (xb_dev * 40.0 + ab_dev * 40.0)))
                if cv >= i_conf
                    xc_len = math.abs(x_p - c_p)
                    d1 = bullish ? c_p - xc_len * 0.886 : c_p + xc_len * 0.886
                    cd_r = f_nearest_cd(ab_bc, 2.240, 2.000, 1.618, 1.618, 1.618)
                    d2 = bullish ? c_p - bc * cd_r : c_p + bc * cd_r
                    dd = math.abs(cur_p - d1) / math.max(math.abs(d1), 1e-10) * 100
                    if dd <= i_ddist
                        [cl, ch, ok] = f_prz(d1, d2, x_p, bullish, false)
                        if ok
                            found := true
                            pname := "Shark"
                            pconf := cv
                            plo   := cl
                            phi   := ch

        // CYPHER — _BC_TO_CD: {0.382:1.272, 0.500:1.000, 0.618:0.786, 0.786:0.786, 0.886:0.786}
        if i_cypher and not found
            c_ex = bullish ? c_p > a_p : c_p < a_p
            if c_ex and f_in(xa_ab, 0.350, 0.650) and f_in(xc_prj, 1.200, 1.500)
                xc_id  = (1.200 + 1.500) / 2.0
                xa_id  = (0.350 + 0.650) / 2.0
                xc_dev = math.abs(xc_prj - xc_id) / math.max(xc_id, 1e-10)
                xa_dev = math.abs(xa_ab  - xa_id) / math.max(xa_id, 1e-10)
                cv = math.max(0.0, math.min(100.0, 100.0 - (xa_dev * 40.0 + xc_dev * 40.0)))
                if cv >= i_conf
                    xc_len = math.abs(x_p - c_p)
                    d1 = bullish ? c_p - xc_len * 0.786 : c_p + xc_len * 0.786
                    cd_r = f_nearest_cd(ab_bc, 1.272, 1.000, 0.786, 0.786, 0.786)
                    d2 = bullish ? c_p - bc * cd_r : c_p + bc * cd_r
                    in_cy = cur_p >= math.min(d1, d2) and cur_p <= math.max(d1, d2)
                    dd = in_cy ? 0.0 : math.abs(cur_p - d1) / math.max(math.abs(d1), 1e-10) * 100
                    if dd <= i_ddist
                        [cl, ch, ok] = f_prz(d1, d2, x_p, bullish, true)
                        if ok
                            found := true
                            pname := "Cypher"
                            pconf := cv
                            plo   := cl
                            phi   := ch

    [found, pname, pconf, plo, phi]

                cb_ = array.get(zb, idx + 3)

                ct  = array.get(zt, idx + 3)

                bull_struct = xt == -1 and at == 1 and bt == -1 and ct == 1
                bear_struct = xt == 1  and at == -1 and bt == 1  and ct == -1

                if bull_struct
                    [fnd, pname, pconf, plo, phi] = f_check(xp, ap, bp, cp, true, close)
                    if fnd
                        lc    = C_BULL
                        pc_bg = C_PRZ_B_BG
                        cs    = str.tostring(math.round(pconf, 1))
                        array.push(h_labels, label.new(cb_, cp, pname + "\nBULL %" + cs, style=label.style_label_up, color=color.new(lc, 10), textcolor=color.white, size=size.small))
                        array.push(h_labels, label.new(xb_, xp, "X", style=label.style_label_up,  color=color.new(lc, 70), textcolor=lc, size=size.tiny))
                        array.push(h_labels, label.new(ab_, ap, "A", style=label.style_label_down, color=color.new(lc, 70), textcolor=lc, size=size.tiny))
                        array.push(h_labels, label.new(bb_, bp, "B", style=label.style_label_up,   color=color.new(lc, 70), textcolor=lc, size=size.tiny))
                        array.push(h_labels, label.new(cb_, cp, "C", style=label.style_label_down, color=color.new(lc, 70), textcolor=lc, size=size.tiny))
                        array.push(h_lines, line.new(xb_, xp, ab_, ap, color=lc, width=2))
                        array.push(h_lines, line.new(ab_, ap, bb_, bp, color=lc, width=2))
                        array.push(h_lines, line.new(bb_, bp, cb_, cp, color=lc, width=2))
                        if not na(plo) and not na(phi)
                            array.push(h_boxes,  box.new(cb_, phi, cb_ + i_prz_ext, plo, bgcolor=pc_bg, border_color=lc, border_width=1))
                            array.push(h_labels, label.new(cb_ + i_prz_ext, (plo + phi) / 2, "PRZ\n" + str.tostring(phi, "#.####") + "\n|\n" + str.tostring(plo, "#.####"), style=label.style_label_left, color=pc_bg, textcolor=lc, size=size.tiny))

                if bear_struct
                    [fnd, pname, pconf, plo, phi] = f_check(xp, ap, bp, cp, false, close)
                    if fnd
                        lc    = C_BEAR
                        pc_bg = C_PRZ_R_BG
                        cs    = str.tostring(math.round(pconf, 1))

                        array.push(h_labels, label.new(xb_, xp, "X", style=label.style_label_down, color=color.new(lc, 70), textcolor=lc, size=size.tiny))

// ─── Candlestick ──────────────────────────────────────────────────────────────
cs_o3 = open[2]
cs_h3 = high[2]
cs_l3 = low[2]
cs_c3 = close[2]
cs_o2 = open[1]
cs_h2 = high[1]
cs_l2 = low[1]
cs_c2 = close[1]
cs_o1 = open
cs_h1 = high
cs_l1 = low
cs_c1 = close
cs_b3 = math.abs(cs_c3 - cs_o3)
cs_r3 = cs_h3 - cs_l3
cs_b2 = math.abs(cs_c2 - cs_o2)
cs_r2 = cs_h2 - cs_l2
cs_b1 = math.abs(cs_c1 - cs_o1)
cs_r1 = cs_h1 - cs_l1
ptol  = 0.20

k_br3    = cs_r3 > 0 ? cs_b3 / cs_r3 : 0.0
k_br2    = cs_r2 > 0 ? cs_b2 / cs_r2 : 0.0
k_br1    = cs_r1 > 0 ? cs_b1 / cs_r1 : 0.0
k_br_min = k_br3 >= i_btr_min and k_br2 >= i_btr_min and k_br1 >= i_btr_min
bmax     = math.max(cs_b3, math.max(cs_b2, cs_b1))
bmin     = math.min(cs_b3, math.min(cs_b2, cs_b1))
k_ratio  = bmin > 0 and bmax / bmin <= 2.0
k_minbod = cs_b3 >= atr14 * 0.15 and cs_b2 >= atr14 * 0.15 and cs_b1 >= atr14 * 0.15

k_open_ok(o_cur, o_prev, c_prev, b_prev) =>
    lo_ok = o_cur >= math.min(o_prev, c_prev) - b_prev * ptol
    hi_ok = o_cur <= math.max(o_prev, c_prev) + b_prev * ptol
    lo_ok and hi_ok

tws_c1 = cs_c3 > cs_o3 and cs_c2 > cs_o2 and cs_c1 > cs_o1
tws_c2 = k_br_min and cs_c2 > cs_c3 and cs_c1 > cs_c2
tws_c3 = k_open_ok(cs_o2, cs_o3, cs_c3, cs_b3)
tws_c4 = k_open_ok(cs_o1, cs_o2, cs_c2, cs_b2)
tws_c5 = (cs_h3 - math.max(cs_o3, cs_c3)) <= cs_b3 * 0.40
tws_c6 = (cs_h2 - math.max(cs_o2, cs_c2)) <= cs_b2 * 0.40
tws_c7 = (cs_h1 - math.max(cs_o1, cs_c1)) <= cs_b1 * 0.40
tws    = i_tws and tws_c1 and tws_c2 and tws_c3 and tws_c4 and tws_c5 and tws_c6 and tws_c7 and k_ratio and k_minbod

tbc_c1 = cs_c3 < cs_o3 and cs_c2 < cs_o2 and cs_c1 < cs_o1
tbc_c2 = k_br_min and cs_c2 < cs_c3 and cs_c1 < cs_c2
tbc_c3 = k_open_ok(cs_o2, cs_o3, cs_c3, cs_b3)
tbc_c4 = k_open_ok(cs_o1, cs_o2, cs_c2, cs_b2)
tbc_c5 = (math.min(cs_o3, cs_c3) - cs_l3) <= cs_b3 * 0.40
tbc_c6 = (math.min(cs_o2, cs_c2) - cs_l2) <= cs_b2 * 0.40
tbc_c7 = (math.min(cs_o1, cs_c1) - cs_l1) <= cs_b1 * 0.40
tbc    = i_tws and tbc_c1 and tbc_c2 and tbc_c3 and tbc_c4 and tbc_c5 and tbc_c6 and tbc_c7 and k_ratio and k_minbod

plotshape(tws, "Three White Soldiers", shape.triangleup,   location.belowbar, color.new(color.lime, 0), size=size.small, text="3WS")
plotshape(tbc, "Three Black Crows",    shape.triangledown, location.abovebar, color.new(color.red,  0), size=size.small, text="3BC")
alertcondition(tws, "Three White Soldiers", "THREE WHITE SOLDIERS!")
alertcondition(tbc, "Three Black Crows",    "THREE BLACK CROWS!")

// ─── Horseshoe ────────────────────────────────────────────────────────────────
m5o = open[4]
m5h = high[4]
m5l = low[4]
m5c = close[4]
m4o = open[3]
m4h = high[3]
m4l = low[3]
m4c = close[3]
m3o = open[2]
m3h = high[2]
m3l = low[2]
m3c = close[2]
m2o = open[1]
m2h = high[1]
m2l = low[1]
m2c = close[1]

m3b = math.abs(m3c - m3o)

hs_b2 = m3l <= m4l and m3l <= m2l and m2c < m2o and close > open

hs_r1 = m5c < m5o and m4c > m4o and m5b > 0 and m3b <= m5b * 0.50

hs_r4 = math.abs(m4c - m2c) / math.max(math.abs(m5c - m3c), 1e-10) < 0.70 and m5b >= atr14 * 0.20
